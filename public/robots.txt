# Live Spectate pages carry the names the scorer typed, so they stay out of
# every index. Under RFC 9309 a crawler obeys exactly one group — the most
# specific one matching its token — so the Disallows are repeated in every
# group rather than parked in a second "User-agent: *" record that named
# crawlers would never read.
User-agent: *
Allow: /
Disallow: /w/
Disallow: /watch
Disallow: /score
Content-Signal: ai-train=no, search=yes, ai-input=yes
